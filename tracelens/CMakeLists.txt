cmake_minimum_required(VERSION 3.22)
project(TraceLens VERSION 1.0.0 LANGUAGES CXX) # x-release-please-version
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TRACELENS_BUILD_GUI "Build the Qt Widgets investigator" ON)
option(TRACELENS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(TRACELENS_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
configure_file(include/tracelens/version.hpp.in generated/tracelens/version.hpp @ONLY)
add_library(tracelens_core src/core.cpp src/sha256.cpp)
target_include_directories(tracelens_core PUBLIC include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_options(tracelens_core PRIVATE -Wall -Wextra -Wpedantic)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_library(tracelens_adapter src/adapter.cpp)
target_link_libraries(tracelens_adapter PUBLIC tracelens_core Qt6::Core)
target_include_directories(tracelens_adapter PUBLIC src)
add_executable(tracelens src/cli.cpp)
target_link_libraries(tracelens PRIVATE tracelens_adapter)
install(TARGETS tracelens RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(TRACELENS_BUILD_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent)
  add_executable(tracelens-gui src/gui.cpp)
  target_link_libraries(tracelens-gui PRIVATE tracelens_adapter Qt6::Widgets Qt6::Concurrent)
  install(TARGETS tracelens-gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES packaging/tracelens.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES packaging/tracelens.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
install(DIRECTORY schemas/ DESTINATION ${CMAKE_INSTALL_DATADIR}/tracelens/schemas)
install(FILES README.md CHANGELOG.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
if(BUILD_TESTING)
  add_executable(tracelens-core-test tests/core_test.cpp)
  target_link_libraries(tracelens-core-test PRIVATE tracelens_core)
  add_test(NAME tracelens-core COMMAND tracelens-core-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME tracelens-cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:tracelens> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)
  if(TRACELENS_BUILD_GUI)
    add_test(NAME tracelens-gui-smoke COMMAND tracelens-gui --smoke-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/mixed.strace)
    set_tests_properties(tracelens-gui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  endif()
endif()
set(CPACK_PACKAGE_NAME tracelens)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR TGZ)
include(CPack)
